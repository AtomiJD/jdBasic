[TXTREADER$]
TXTREADER$(filename$, [encoding$]) -> string$
Reads the entire contents of a text file into a string. If `encoding$` is given, the file's bytes are decoded from that codepage to UTF-8. Default (no arg / "" / "utf-8") is byte-pass-through. Supported: "cp1252"/"windows-1252"/"latin1", "cp850", "cp437", "cp1250", "cp1251", "cp936", "cp932". Codepage decode is Windows-only.

[TXTWRITER]
TXTWRITER filename$, content$, [append], [encoding$]
Writes a string to a text file. The optional 3rd arg `append` (TRUE = append, default FALSE = overwrite). The optional 4th arg `encoding$` re-encodes the UTF-8 jdBasic string to the named codepage on disk; same name set as TXTREADER$. Binary mode, no newline translation. Roundtrip example:
    DIM src$ = TXTREADER$("export.bas", "cp1252")
    TXTWRITER "export.bas", src$, FALSE, "cp1252"

Syntax: FORM.TIMER(frm, name$, interval_ms) -> handle
Description: Add a timer that fires NAME_TICK every interval_ms; 0 creates it disabled. Change with FORM.SET(h, "INTERVAL", ms).

[FORM.SET]
Syntax: FORM.SET(handle, prop$, value)
Description: Set a property: TEXT, ENABLED, VISIBLE, CHECKED, ITEMS (array), ADDITEM, CLEAR, SELINDEX, FOCUS, X, Y, WIDTH, HEIGHT, INTERVAL (timer), VALUE (TRUE clicks a button).

[FORM.GET]
Syntax: FORM.GET(handle, prop$) -> value
Description: Read a property: TEXT, ENABLED, VISIBLE, CHECKED, SELINDEX, SELTEXT, COUNT, NAME, KIND, HWND, X, Y, WIDTH, HEIGHT.

[FORM.RUN]
Syntax: FORM.RUN(frm)
Description: Show the form and run the message loop until every form is closed. Fires NAME_LOAD first; handlers run on the main thread.

[FORM.SHOW]
Syntax: FORM.SHOW(frm)
Description: Show a (secondary) form without blocking and fire its NAME_LOAD event.

[FORM.DOEVENTS]
Syntax: FORM.DOEVENTS() -> bool
Description: Pump pending window messages once; TRUE while any form is open. For DO...LOOP style programs instead of FORM.RUN.

[FORM.CLOSE]
Syntax: FORM.CLOSE(frm)
Description: Close a form. When the last form closes, FORM.RUN returns.

[MSGBOX]
Syntax: MSGBOX(text$, [flags], [title$]) -> button
Description: Show a message box. Flags follow VB6/Win32: 1=OKCancel 4=YesNo 16=Critical 32=Question 48=Exclamation 64=Information. Returns 1=OK 2=Cancel 6=Yes 7=No.

[INPUTBOX$]
Syntax: INPUTBOX$(prompt$, [title$], [default$]) -> string
Description: Modal text input dialog. Returns the entered text, or "" on Cancel.

[FORM.LOAD]
Syntax: FORM.LOAD(path$) -> handle
Description: Instantiate a .jdform file (JSON layout) and auto-bind every SUB named CONTROL_EVENT in the program. Relative paths resolve against the script directory.

[FORM.FIND]
Syntax: FORM.FIND(frm, name$) -> handle
Description: Look up a control of a form by name (case-insensitive); 0 if absent. Companion to FORM.LOAD for code-behind property access.

[FORM.MENU]
Syntax: FORM.MENU(frm, spec)
Description: Build or replace the form's menu bar. Spec = array of maps: text (& = Alt key, "-" = separator), items (submenu), name (clickable -> NAME_CLICK), key ("Ctrl+O", "F5" - real accelerator). Items support FORM.SET ENABLED/CHECKED/TEXT/VALUE.

[FORM.MDI]
Syntax: FORM.MDI(title$, width, height, [name$]) -> handle
Description: Create an MDI frame window (multi-document parent). Combine with FORM.MENU and FORM.CHILD; run with FORM.RUN.

[FORM.CHILD]
Syntax: FORM.CHILD(frame, title$, width, height, [name$]) -> handle
Description: Create an MDI child document inside a frame. Hosts controls like a form; closing children never ends the program, closing the frame does.

[FORM.TOOLBAR]
Syntax: FORM.TOOLBAR(frm, name$, spec) -> handle
Description: Flat icon toolbar along the top (one per form). Spec = array of maps: name (-> NAME_CLICK), text (tooltip), icon (NEW/OPEN/SAVE/CUT/COPY/PASTE/UNDO/REDO/DELETE/PRINT/FIND/REPLACE/PROPERTIES/HELP), check (toggle); "-" = separator. Buttons take FORM.SET ENABLED/CHECKED/VALUE.

[FORM.STATUSBAR]
Syntax: FORM.STATUSBAR(frm, name$, [part_widths]) -> handle
Description: Status bar along the bottom (one per form). part_widths = array of logical widths, -1 stretches the last part; omitted = one part. SET TEXT takes a string (part 0) or an array (all parts).

[FORM.LINE]
Syntax: FORM.LINE(frm, name$, x, y, length, [vertical]) -> handle
Description: Etched divider line, horizontal by default.

[FORM.SHAPE]
Syntax: FORM.SHAPE(frm, name$, x, y, w, h, [rgb]) -> handle
Description: Filled rectangle. COLOR property (0xRRGGBB) via FORM.SET/GET.

[FORM.PICTURE]
Syntax: FORM.PICTURE(frm, name$, path$, x, y, w, h) -> handle
Description: Image control (PNG/JPG/BMP/GIF), scaled to the control. Swap via FORM.SET PICTURE.

[FORM.PROGRESS]
Syntax: FORM.PROGRESS(frm, name$, x, y, w, h) -> handle
Description: Progress bar; VALUE and MAX properties (default 0..100).

[FORM.SLIDER]
Syntax: FORM.SLIDER(frm, name$, x, y, w, h, [max], [vertical]) -> handle
Description: Trackbar; VALUE/MIN/MAX; fires NAME_CHANGE with value while dragging.

[FORM.UPDOWN]
Syntax: FORM.UPDOWN(frm, name$, x, y, w, h, [min], [max]) -> handle
Description: Numeric edit with spin arrows; VALUE/MIN/MAX/TEXT; fires NAME_CHANGE.

[FORM.LISTVIEW]
Syntax: FORM.LISTVIEW(frm, name$, x, y, w, h) -> handle
Description: Report list with columns and grid lines. COLUMNS (array of maps or strings), ITEMS (array of row-arrays), ADDITEM, CLEAR, SELINDEX, SELTEXT, COUNT; cells via FORM.GET(h, "CELL", row, col). Fires NAME_CLICK/NAME_DBLCLICK.

[FORM.TREEVIEW]
Syntax: FORM.TREEVIEW(frm, name$, x, y, w, h) -> handle
Description: Hierarchy control; add nodes with FORM.NODE. SELNODE/SELTEXT/COUNT/EXPAND/CLEAR; fires NAME_CLICK on selection change.

[FORM.NODE]
Syntax: FORM.NODE(tree, parent, text$) -> node
Description: Insert a tree node under parent (0 = root level); returns the node id.

[FORM.TABS]
Syntax: FORM.TABS(frm, name$, x, y, w, h, titles) -> handle
Description: Tab strip from an array of titles. Assign controls to pages with FORM.TABPAGE; SELINDEX/COUNT; fires NAME_CHANGE.

[FORM.TABPAGE]
Syntax: FORM.TABPAGE(tab, page, ctl)
Description: Put a control on a tab page - its visibility follows the selected tab.

[FORM.DATETIME]
Syntax: FORM.DATETIME(frm, name$, x, y, w, h) -> handle
Description: Date picker. VALUE is an epoch (use CVDATE/YEAR/MONTH/DAY); fires NAME_CHANGE.

[FORM.RICHTEXT]
Syntax: FORM.RICHTEXT(frm, name$, x, y, w, h) -> handle
Description: Rich-edit box; TEXT get/set; fires NAME_CHANGE.

[FILEOPEN$]

[FILESAVE$]
